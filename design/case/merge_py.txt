import FreeCAD as App
def merge():
    c = App.getDocument('crown')
    m = App.getDocument('master')
    src = m.getObject('Params')
    dst = c.getObject('Crown')
    row = 1
    for r in range(1, 45):
        a = src.getAlias('B%d' % r)
        if not a:
            continue
        if dst.getAlias('B%d' % row):
            row += 1; continue
        dst.set('A%d' % row, a)
        dst.set('B%d' % row, str(src.getContents('B%d' % r)))
        dst.setAlias('B%d' % row, a)
        try:
            dst.set('C%d' % row, src.get('C%d' % r))
        except Exception:
            pass
        print("copied", a)
        row += 1
    # crown depth belongs in Params, since the case needs it
    dst.set('A%d' % row, 'CrownD')
    dst.set('B%d' % row, '=3 mm')
    dst.setAlias('B%d' % row, 'CrownD')
    c.recompute()
    sh = c.getObject('Spreadsheet')
    sh.set(sh.getCellFromAlias('D'), '=<<Params>>.CrownD')
    sh.set(sh.getCellFromAlias('BoreD'), '=<<Params>>.PinD + <<Params>>.ClrSlip')
    c.recompute()
    print("D =", sh.get('D'), " BoreD =", sh.get('BoreD'))
merge()
